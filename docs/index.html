Quick Start

Install HeadlessUI via loadstring in your script executor:

local HeadlessUI = loadstring(game:HttpGet("https://raw.githubusercontent.com/User/Repo/main/HeadlessUI.lua"))()
local UI = HeadlessUI.new()

local Window = UI:CreateWindow({Title = "RBXTOOLS"})
local Tab = Window:AddTab("Combat")

Tab:AddToggle("Aimbot", function(state)
    print("Aimbot is now:", state)
end)

UI:Notify("Success", "HeadlessUI Loaded!")
